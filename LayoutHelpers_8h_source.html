64struct StaffKeyHash
65{
70 size_t operator()(const StaffKey& k) const noexcept
71 {
72 size_t h1 = std::hash<std::string>{}(k.partId);
73 size_t h2 = std::hash<int>{}(k.staffNo);
74 return h1 ^ (h2 + 0x9e3779b9u + (h1 << 6) + (h1 >> 2));
75 }
76};

93[[nodiscard]] inline std::optional<LayoutStaffKeySet>
94analyzeLayoutStaffVoices(const layout::Staff& staff)
95{
96 const auto sources = staff.sources();
97 if (sources.empty()) {
98 return std::nullopt;
99 }
100
101 struct KeyState
102 {
103 size_t count = 0;
104 bool anyMissingVoice = false;
105 std::unordered_set<std::string> voices;
106 };
107
108 std::unordered_map<StaffKey, KeyState, StaffKeyHash> stateByKey;
109 stateByKey.reserve(sources.size());
110
111 for (const auto src : sources) {
112 const std::string partId = src.part();
113 if (partId.empty()) {
114 return std::nullopt;
115 }
116
117 const StaffKey key{partId, src.staff()};
118 auto& st = stateByKey[key];
119 ++st.count;
120
121 const auto v = src.voice();
122 if (!v) {
123 st.anyMissingVoice = true;
124 continue;
125 }
126
127 if (v->empty()) {
128 return std::nullopt;
129 }
130
131 if (!st.voices.emplace(*v).second) {
132 return std::nullopt; // duplicate voice for same StaffKey
133 }
134 }
135
136 // Enforce per-StaffKey semantic rule.
137 for (const auto& kv : stateByKey) {
138 const KeyState& st = kv.second;
139 if (st.count > 1) {
140 if (st.anyMissingVoice || st.voices.size() != st.count) {
141 return std::nullopt;
142 }
143 }
144 }
145
146 LayoutStaffKeySet result;
147 result.reserve(stateByKey.size());
148 for (const auto& kv : stateByKey) {
149 result.insert(kv.first);
150 }
151
152 return result;
153}

164[[nodiscard]] inline std::optional<std::vector<layout::Staff>>
165flattenLayoutStaves(const Layout& layout)
166{
167 auto content = layout.content();
168 std::vector<layout::Staff> result;
169 result.reserve(content.size()); // lower bound; groups may expand further
170
171 const auto walk = [&](auto&& self, const layout::LayoutContent& content) -> std::optional<bool> {
172 for (auto elem : content) {
173 if (elem.type() == layout::Group::ContentTypeValue) {
174 layout::Group g = elem.get<layout::Group>();
175 auto ok = self(self, g.content());
176 if (!ok) {
177 return std::nullopt;
178 }
179 } else if (elem.type() == layout::Staff::ContentTypeValue) {
180 result.push_back(elem.get<layout::Staff>());
181 } else {
182 return std::nullopt;
183 }
184 }
185 return true;
186 };
187
188 if (!walk(walk, content)) {
189 return std::nullopt;
190 }
191 return result;
192}

345[[nodiscard]] inline std::optional<std::vector<LayoutSpan>>
346buildLayoutSpans(const mnx::Layout& layout)
347{
348 const auto content = layout.content();
349 std::vector<LayoutSpan> spans;
350 spans.reserve(content.size()); // lower bound
351
352 size_t staffIndex = 0;
353 size_t encounter = 0; // stable tiebreaker
354
355 struct SortKey { size_t start, depth, encounter; };
356
357 struct TaggedSpan {
358 LayoutSpan span;
359 SortKey key;
360 };
361
362 std::vector<TaggedSpan> tagged;
363 tagged.reserve(content.size());
364
365 // Return value:
366 // - outer std::optional: hard failure (unsupported element) if empty
367 // - inner std::optional: no staves in subtree if empty; otherwise first/last staff indices
368 const auto walk =
369 [&](auto&& self, const layout::LayoutContent& arr, size_t depth)
370 -> std::optional<std::optional<std::pair<size_t,size_t>>>
371 {
372 std::optional<size_t> first;
373 std::optional<size_t> last;
374
375 for (auto elem : arr) {
376 if (elem.type() == layout::Staff::ContentTypeValue) {
377 layout::Staff s = elem.get<layout::Staff>();
378
379 const size_t i = staffIndex++;
380
381 LayoutSpan span;
382 span.kind = LayoutSpan::Kind::Staff;
383 span.depth = depth + 1; // staff spans must be at the deepest depth
384 span.startIndex = i;
385 span.endIndex = i;
386 span.symbol = s.symbol();
387 span.label = s.label();
388 if (const auto labelRef = s.labelRef()) {
389 LayoutSpanLabelRef value(*labelRef);
390 if (!value.target && !s.sources().empty()) {
391 value.target = s.sources()[0].part(); // MNX fallback for staves
392 }
393 span.labelRef = std::move(value);
394 }
395 span.barlineOverride = StaffGroupBarlineOverride::None;
396 span.sources = util::analyzeLayoutStaffVoices(s);
397
398 tagged.push_back({ std::move(span), SortKey{i, depth + 1, encounter++} });
399
400 first = first.value_or(i);
401 last = i;
402 } else if (elem.type() == layout::Group::ContentTypeValue) {
403 layout::Group g = elem.get<layout::Group>();
404
405 auto childRange = self(self, g.content(), depth + 1);
406 if (!childRange) {
407 return std::nullopt; // hard failure
408 }
409 if (!*childRange) {
410 continue; // skip groups with no staves anywhere in their subtree
411 }
412 const auto [cFirst, cLast] = **childRange;
413
414 LayoutSpan span;
415 span.kind = LayoutSpan::Kind::Group;
416 span.depth = depth;
417 span.startIndex = cFirst;
418 span.endIndex = cLast;
419 span.symbol = g.symbol();
420 span.label = g.label();
421 if (const auto labelRef = g.labelRef()) {
422 span.labelRef = LayoutSpanLabelRef(*labelRef);
423 }
424 span.barlineOverride = g.calcBarlineOverride();
425
426 tagged.push_back({ std::move(span), SortKey{cFirst, depth, encounter++} });
427
428 first = first.value_or(cFirst);
429 last = cLast;
430 } else {
431 return std::nullopt; // unsupported content element
432 }
433 }
434
435 if (!first || !last) {
436 return std::optional<std::pair<size_t,size_t>>{}; // success, but no staves
437 }
438 return std::make_pair(*first, *last);
439 };
440
441 auto rootRange = walk(walk, content, /*depth*/0);
442 if (!rootRange) {
443 return std::nullopt;
444 }
445 // Empty root content is allowed; it simply produces an empty spans vector.
446
447 std::stable_sort(tagged.begin(), tagged.end(),
448 [](const TaggedSpan& a, const TaggedSpan& b)
449 {
450 if (a.key.start != b.key.start) return a.key.start < b.key.start;
451 if (a.key.depth != b.key.depth) return a.key.depth < b.key.depth;
452 return a.key.encounter < b.key.encounter;
453 });
454
455 spans.reserve(tagged.size());
456 for (auto& t : tagged) spans.push_back(std::move(t.span));
457 return spans;
458}

481[[nodiscard]] inline std::vector<LayoutSpan>
482buildDefaultLayoutSpans(const Array<Part>& parts)
483{
484 std::vector<LayoutSpan> result;
485 size_t staffIdx = 0;
486
487 for (const auto& part : parts) {
488 const size_t numStaves = static_cast<size_t>(part.staves());
489 if (numStaves == 0) {
490 continue;
491 }
492 const auto applyPartLabel = [&](LayoutSpan& span) {
493 if (const auto partId = part.id()) {
494 LayoutSpanLabelRef labelRef;
495 labelRef.ref = LabelRefType::Auto;
496 labelRef.target = partId;
497 span.labelRef = std::move(labelRef);
498 } else {
499 span.label = part.name();
500 }
501 };
502 size_t staffDepth = 1;
503 bool staffNameNeeded = true;
504 if (numStaves > 1) {
505 LayoutSpan groupSpan;
506 groupSpan.depth = 0;
507 groupSpan.kind = LayoutSpan::Kind::Group;
508 groupSpan.symbol = LayoutSymbol::Brace;
509 groupSpan.startIndex = staffIdx;
510 groupSpan.endIndex = staffIdx + numStaves - 1;
511 applyPartLabel(groupSpan);
512 groupSpan.barlineOverride = StaffGroupBarlineOverride::Unified;
513 staffNameNeeded = false;
514 result.emplace_back(std::move(groupSpan));
515 }
516 for (size_t x = 0; x < numStaves; x++) {
517 LayoutSpan staffSpan;
518 staffSpan.depth = staffDepth;
519 staffSpan.kind = LayoutSpan::Kind::Staff;
520 staffSpan.startIndex = staffIdx;
521 staffSpan.endIndex = staffIdx;
522 staffSpan.symbol = LayoutSymbol::NoSymbol;
523 staffSpan.barlineOverride = StaffGroupBarlineOverride::None;
524 if (staffNameNeeded) {
525 applyPartLabel(staffSpan);
526 }
527 result.emplace_back(std::move(staffSpan));
528 staffIdx++;
529 }
530 }
531
532 return result;
533}
