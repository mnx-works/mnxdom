74struct [[nodiscard]] FractionValue
75{
77 using NumType = unsigned;
78 static_assert(std::is_integral<NumType>::value, "FractionValue::NumType must be an integral type");
79
80private:
81 NumType m_num = 0;
82 NumType m_den = 1;
83
84 template<typename T = NumType,
85 typename std::enable_if_t<std::is_signed<T>::value, int> = 0>
86 constexpr void normalizeSign()
87 {
88 if (m_den < 0) {
89 m_den = -m_den;
90 m_num = -m_num;
91 }
92 }
93
94 // Unsigned case: no-op, never even sees m_den < 0
95 template<typename T = NumType,
96 typename std::enable_if_t<!std::is_signed<T>::value, int> = 0>
97 constexpr void normalizeSign() {}
98
99public:
103 constexpr FractionValue() = default;
104
117 constexpr FractionValue(NumType num, NumType den)
118 : m_num(num), m_den(den)
119 {
120 if (m_den == 0) {
121 throw std::invalid_argument("FractionValue: denominator must not be zero.");
122 }
123 normalizeSign();
124 }
125
131 constexpr FractionValue(NumType value) : m_num(value), m_den(1) {}
132
134 [[nodiscard]] constexpr NumType numerator() const noexcept { return m_num; }
135
137 [[nodiscard]] constexpr NumType denominator() const noexcept { return m_den; }
138
143 [[nodiscard]] constexpr NumType quotient() const
144 {
145 return m_num / m_den;
146 }
147
152 constexpr FractionValue remainder() const
153 {
154 FractionValue result;
155 result.m_num = m_num % m_den;
156 result.m_den = m_den;
157 return result;
158 }
159
161 static constexpr FractionValue max() noexcept
162 {
163 return FractionValue((std::numeric_limits<NumType>::max)());
164 }
165
167 [[nodiscard]] constexpr double toDouble() const {
168 return double(m_num) / double(m_den);
169 }
170
179 constexpr FractionValue& operator+=(const FractionValue& rhs)
180 {
181 // a/b + c/d = (ad + bc)/bd
182 m_num = m_num * rhs.m_den + rhs.m_num * m_den;
183 m_den = m_den * rhs.m_den;
184 reduce();
185 return *this;
186 }
187
196 constexpr FractionValue& operator-=(const FractionValue& rhs)
197 {
198 // a/b - c/d = (ad - bc)/bd
199 m_num = m_num * rhs.m_den - rhs.m_num * m_den;
200 m_den = m_den * rhs.m_den;
201 reduce();
202 return *this;
203 }
204
213 constexpr FractionValue& operator*=(const FractionValue& rhs)
214 {
215 m_num = m_num * rhs.m_num;
216 m_den = m_den * rhs.m_den;
217 reduce();
218 return *this;
219 }
220
231 constexpr FractionValue& operator/=(const FractionValue& rhs)
232 {
233 if (rhs.m_num == 0) {
234 throw std::invalid_argument("Division by zero FractionValue.");
235 }
236 m_num = m_num * rhs.m_den;
237 m_den = m_den * rhs.m_num;
238 reduce();
239 return *this;
240 }
241
245 constexpr void reduce()
246 {
247 const NumType g = std::gcd(m_num, m_den);
248 if (g > 1) {
249 m_num /= g;
250 m_den /= g;
251 }
252 normalizeSign();
253 }
254
258 constexpr FractionValue reduced() const
259 {
260 auto result = *this;
261 result.reduce();
262 return result;
263 }
264
295 constexpr bool expressWithDenominator(NumType targetDenominator)
296 {
297 if (targetDenominator == 0) {
298 // Cannot express anything with denominator 0.
299 return false;
300 }
301
302 // Zero fraction: 0/d is expressible as 0/targetDenominator.
303 if (m_num == 0) {
304 m_den = targetDenominator;
305 return true;
306 }
307
308 // Reduce the fraction logically: m_num/m_den -> (numRed/denRed).
309 // We don't actually have to store the reduced form; just use it to
310 // check compatibility and compute the new numerator.
311 const NumType g = std::gcd(m_num, m_den);
312 const NumType denRed = m_den / g;
313 const NumType numRed = m_num / g; // sign preserved
314
315 // For an integer representation with the requested denominator to exist
316 // (while preserving the value), the reduced denominator must divide
317 // targetDenominator exactly.
318 if (targetDenominator % denRed != 0) {
319 return false; // No integer scaling factor exists.
320 }
321
322 const NumType factor = targetDenominator / denRed;
323
324 // Apply the scaling to the reduced numerator and store the new form.
325 m_num = numRed * factor;
326 m_den = targetDenominator;
327
328 normalizeSign();
329
330 return true;
331 }
