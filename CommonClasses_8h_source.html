76struct [[nodiscard]] FractionValue
77{
79 using NumType = unsigned;
80 static_assert(std::is_integral<NumType>::value, "FractionValue::NumType must be an integral type");
81
82private:
83 NumType m_num = 0;
84 NumType m_den = 1;
85
86 template<typename T = NumType,
87 typename std::enable_if_t<std::is_signed<T>::value, int> = 0>
88 constexpr void normalizeSign()
89 {
90 if (m_den < 0) {
91 m_den = -m_den;
92 m_num = -m_num;
93 }
94 }
95
96 // Unsigned case: no-op, never even sees m_den < 0
97 template<typename T = NumType,
98 typename std::enable_if_t<!std::is_signed<T>::value, int> = 0>
99 constexpr void normalizeSign() {}
100
101public:
105 constexpr FractionValue() = default;
106
119 constexpr FractionValue(NumType num, NumType den)
120 : m_num(num), m_den(den)
121 {
122 if (m_den == 0) {
123 throw std::invalid_argument("FractionValue: denominator must not be zero.");
124 }
125 normalizeSign();
126 }
127
133 constexpr FractionValue(NumType value) : m_num(value), m_den(1) {}
134
136 [[nodiscard]] constexpr NumType numerator() const noexcept { return m_num; }
137
139 [[nodiscard]] constexpr NumType denominator() const noexcept { return m_den; }
140
145 [[nodiscard]] constexpr NumType quotient() const
146 {
147 return m_num / m_den;
148 }
149
154 constexpr FractionValue remainder() const
155 {
156 FractionValue result;
157 result.m_num = m_num % m_den;
158 result.m_den = m_den;
159 return result;
160 }
161
163 static constexpr FractionValue max() noexcept
164 {
165 return FractionValue((std::numeric_limits<NumType>::max)());
166 }
167
169 [[nodiscard]] constexpr double toDouble() const {
170 return double(m_num) / double(m_den);
171 }
172
181 constexpr FractionValue& operator+=(const FractionValue& rhs)
182 {
183 // a/b + c/d = (ad + bc)/bd
184 m_num = m_num * rhs.m_den + rhs.m_num * m_den;
185 m_den = m_den * rhs.m_den;
186 reduce();
187 return *this;
188 }
189
198 constexpr FractionValue& operator-=(const FractionValue& rhs)
199 {
200 // a/b - c/d = (ad - bc)/bd
201 m_num = m_num * rhs.m_den - rhs.m_num * m_den;
202 m_den = m_den * rhs.m_den;
203 reduce();
204 return *this;
205 }
206
215 constexpr FractionValue& operator*=(const FractionValue& rhs)
216 {
217 m_num = m_num * rhs.m_num;
218 m_den = m_den * rhs.m_den;
219 reduce();
220 return *this;
221 }
222
233 constexpr FractionValue& operator/=(const FractionValue& rhs)
234 {
235 if (rhs.m_num == 0) {
236 throw std::invalid_argument("Division by zero FractionValue.");
237 }
238 m_num = m_num * rhs.m_den;
239 m_den = m_den * rhs.m_num;
240 reduce();
241 return *this;
242 }
243
247 constexpr void reduce()
248 {
249 const NumType g = std::gcd(m_num, m_den);
250 if (g > 1) {
251 m_num /= g;
252 m_den /= g;
253 }
254 normalizeSign();
255 }
256
260 constexpr FractionValue reduced() const
261 {
262 auto result = *this;
263 result.reduce();
264 return result;
265 }
266
297 constexpr bool expressWithDenominator(NumType targetDenominator)
298 {
299 if (targetDenominator == 0) {
300 // Cannot express anything with denominator 0.
301 return false;
302 }
303
304 // Zero fraction: 0/d is expressible as 0/targetDenominator.
305 if (m_num == 0) {
306 m_den = targetDenominator;
307 return true;
308 }
309
310 // Reduce the fraction logically: m_num/m_den -> (numRed/denRed).
311 // We don't actually have to store the reduced form; just use it to
312 // check compatibility and compute the new numerator.
313 const NumType g = std::gcd(m_num, m_den);
314 const NumType denRed = m_den / g;
315 const NumType numRed = m_num / g; // sign preserved
316
317 // For an integer representation with the requested denominator to exist
318 // (while preserving the value), the reduced denominator must divide
319 // targetDenominator exactly.
320 if (targetDenominator % denRed != 0) {
321 return false; // No integer scaling factor exists.
322 }
323
324 const NumType factor = targetDenominator / denRed;
325
326 // Apply the scaling to the reduced numerator and store the new form.
327 m_num = numRed * factor;
328 m_den = targetDenominator;
329
330 normalizeSign();
331
332 return true;
333 }

418 throw std::invalid_argument("mnx::Fraction must have exactly 2 elements.");
