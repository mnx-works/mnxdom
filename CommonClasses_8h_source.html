76struct [[nodiscard]] FractionValue
77{
79 using NumType = unsigned;
80 static_assert(std::is_integral<NumType>::value, "FractionValue::NumType must be an integral type");
81
82private:
83 NumType m_num = 0;
84 NumType m_den = 1;
85
86 template<typename T = NumType,
87 typename std::enable_if_t<std::is_signed<T>::value, int> = 0>
88 constexpr void normalizeSign()
89 {
90 if (m_den < 0) {
91 m_den = -m_den;
92 m_num = -m_num;
93 }
94 }
95
96 // Unsigned case: no-op, never even sees m_den < 0
97 template<typename T = NumType,
98 typename std::enable_if_t<!std::is_signed<T>::value, int> = 0>
99 constexpr void normalizeSign() {}
100
101public:
105 constexpr FractionValue() = default;
106
118 constexpr FractionValue(NumType num, NumType den)
119 : m_num(num), m_den(den)
120 {
121 if (m_den == 0) {
122 throw std::invalid_argument("FractionValue: denominator must not be zero.");
123 }
124 normalizeSign();
125 }
126
132 constexpr FractionValue(NumType value) : m_num(value), m_den(1) {}
133
135 [[nodiscard]] constexpr NumType numerator() const noexcept { return m_num; }
136
138 [[nodiscard]] constexpr NumType denominator() const noexcept { return m_den; }
139
144 [[nodiscard]] constexpr NumType quotient() const
145 {
146 return m_num / m_den;
147 }
148
153 constexpr FractionValue remainder() const
154 {
155 FractionValue result;
156 result.m_num = m_num % m_den;
157 result.m_den = m_den;
158 return result;
159 }
160
162 static constexpr FractionValue max() noexcept
163 {
164 return FractionValue((std::numeric_limits<NumType>::max)());
165 }
166
168 [[nodiscard]] constexpr double toDouble() const {
169 return double(m_num) / double(m_den);
170 }
171
180 constexpr FractionValue& operator+=(const FractionValue& rhs)
181 {
182 // a/b + c/d = (ad + bc)/bd
183 m_num = m_num * rhs.m_den + rhs.m_num * m_den;
184 m_den = m_den * rhs.m_den;
185 reduce();
186 return *this;
187 }
188
197 constexpr FractionValue& operator-=(const FractionValue& rhs)
198 {
199 // a/b - c/d = (ad - bc)/bd
200 m_num = m_num * rhs.m_den - rhs.m_num * m_den;
201 m_den = m_den * rhs.m_den;
202 reduce();
203 return *this;
204 }
205
214 constexpr FractionValue& operator*=(const FractionValue& rhs)
215 {
216 m_num = m_num * rhs.m_num;
217 m_den = m_den * rhs.m_den;
218 reduce();
219 return *this;
220 }
221
232 constexpr FractionValue& operator/=(const FractionValue& rhs)
233 {
234 if (rhs.m_num == 0) {
235 throw std::invalid_argument("Division by zero FractionValue.");
236 }
237 m_num = m_num * rhs.m_den;
238 m_den = m_den * rhs.m_num;
239 reduce();
240 return *this;
241 }
242
246 constexpr void reduce()
247 {
248 const NumType g = std::gcd(m_num, m_den);
249 if (g > 1) {
250 m_num /= g;
251 m_den /= g;
252 }
253 normalizeSign();
254 }
255
259 constexpr FractionValue reduced() const
260 {
261 auto result = *this;
262 result.reduce();
263 return result;
264 }
265
296 constexpr bool expressWithDenominator(NumType targetDenominator)
297 {
298 if (targetDenominator == 0) {
299 // Cannot express anything with denominator 0.
300 return false;
301 }
302
303 // Zero fraction: 0/d is expressible as 0/targetDenominator.
304 if (m_num == 0) {
305 m_den = targetDenominator;
306 return true;
307 }
308
309 // Reduce the fraction logically: m_num/m_den -> (numRed/denRed).
310 // We don't actually have to store the reduced form; just use it to
311 // check compatibility and compute the new numerator.
312 const NumType g = std::gcd(m_num, m_den);
313 const NumType denRed = m_den / g;
314 const NumType numRed = m_num / g; // sign preserved
315
316 // For an integer representation with the requested denominator to exist
317 // (while preserving the value), the reduced denominator must divide
318 // targetDenominator exactly.
319 if (targetDenominator % denRed != 0) {
320 return false; // No integer scaling factor exists.
321 }
322
323 const NumType factor = targetDenominator / denRed;
324
325 // Apply the scaling to the reduced numerator and store the new form.
326 m_num = numRed * factor;
327 m_den = targetDenominator;
328
329 normalizeSign();
330
331 return true;
332 }
